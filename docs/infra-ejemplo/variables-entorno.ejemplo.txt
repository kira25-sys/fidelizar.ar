# Example only — copy to the server as .env and fill in every CAMBIAR_ESTO.
# The real .env never enters the repository and is never read by an agent (CLAUDE.md).
#
# .NET reads nested configuration keys with a double underscore as the separator:
# ConnectionStrings__DefaultConnection sets ConnectionStrings:DefaultConnection.

# ---------------------------------------------------------------------------
# Mandatory. The application refuses to start without these.
# ---------------------------------------------------------------------------

# Where the ledger lives. The API fails at startup with a clear message if it is missing,
# rather than throwing on the first request that touches a balance.

# Must match the db service in compose.yml.
POSTGRES_DB=CAMBIAR_ESTO
POSTGRES_USER=CAMBIAR_ESTO
POSTGRES_PASSWORD=CAMBIAR_ESTO

ASPNETCORE_ENVIRONMENT=Production
ASPNETCORE_HTTP_PORTS=8080

# Generate it ON THE SERVER and nowhere else:
#     openssl rand -base64 48

# It is never committed, not even to a development settings file: a key that touches git is in
# the history forever and is burned (ARCHITECTURE §8). The application validates it at startup —
# a missing or under-32-byte key stops the boot with a clear message instead of failing on the
# first login attempt (Jwt:SigningKey bound from this exact variable).
